#最低版本要求
cmake_minimum_required(VERSION 4.0)

#项目名称
project(YCore)

# ===== C++ 标准选择开关 =====
# 另一台只支持 C++17 的机器: 把下面的开关改为 ON
# C++17 模式下, 库内部用 fmt/gsl/内置Expected 替代标准库 expected/format/print/span
# (base_config.hpp 按语言标准自动切换 fmt_ns/span_ns/exp_ns)
option(YCORE_USE_CPP17 "使用 C++17 兼容模式(用 fmt/gsl 替代标准库 expected/format/print/span)" OFF)
if(YCORE_USE_CPP17)
    set(CMAKE_CXX_STANDARD 17)
else()
    set(CMAKE_CXX_STANDARD 23)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(MSVC)
    #设置编译选项
    add_compile_options(/utf-8)
    #MSVC默认定义min/max宏, 会破坏std::min/std::max, 统一禁用
    add_compile_definitions(NOMINMAX)
else()
    #-Wall 开启常用警告
    #-Wextra 开启额外的警告
    #-Wuninitialized 警告未初始化的变量
    #-Wreorder 警告成员初始化顺序和声明顺序不一致
    add_compile_options(-finput-charset=UTF-8 -fexec-charset=UTF-8 
                        -Wuninitialized -Wreorder)
                        #-Wall -Wextra -Wuninitialized -Wreorder)
endif()

set(LIBNAME YCore)

#源文件列表,只能获取当前目录下的
#file(GLOB SRC_FILES ./YCore/*.cpp)

#可以递归获取  CONFIGURE_DEPENDS cmake3.12后自动检查文件变化，之前需要手动删除缓存重新生成
file(GLOB_RECURSE SRC_FILES CONFIGURE_DEPENDS ./YCore/*.cpp)

#message(STATUS "Source files: ${SRC_FILES}")

#生成静态库
add_library(${LIBNAME} STATIC ${SRC_FILES})
#GCC添加实验性质库支持(C++23实验特性, C++17模式不需要)
if(NOT YCORE_USE_CPP17)
    target_link_libraries(${LIBNAME} PUBLIC $<$<CXX_COMPILER_ID:GNU>:stdc++exp>)
endif()   

target_link_directories(${LIBNAME} PUBLIC ${CMAKE_SOURCE_DIR}/YCore/LibAlgorithm/lib)

#target_link_libraries(${LIBNAME} PUBLIC ${CMAKE_SOURCE_DIR}/YCore/LibAlgorithm/lib/libfftw3.a)

target_link_libraries(${LIBNAME} PUBLIC $<IF:$<CXX_COMPILER_ID:MSVC>,fftw3,libfftw3.a>)

#设置包含目录
target_include_directories(${LIBNAME} PUBLIC ./YCore)

#包含format库
add_subdirectory(./YCore/third_party/fmt)
#包含gsl库
add_subdirectory(./YCore/third_party/GSL)

target_link_libraries(${LIBNAME} PRIVATE fmt::fmt Microsoft.GSL::GSL)


add_subdirectory(Test)

add_subdirectory(Test/WASAPITest)
add_subdirectory(Test/ASIOTest)
add_subdirectory(Test/ALGTest)
add_subdirectory(Test/CommonTest)
